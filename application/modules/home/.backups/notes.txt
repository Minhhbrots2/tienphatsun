function loan_calculate(array $opt){
	global $clsISO;
	// $clsISO->print_pre($opt); die();
	// ==========================
    // Input
    // ==========================
    $total_price   = $opt['total_price'];
    $loan_rate     = $opt['loan_rate'];       // %
    $loan_amount   = $total_price * $loan_rate / 100;

    $total_months  = $opt['total_months'];
    $promo_months  = $opt['promo_months'];

    $promo_rate    = $opt['promo_interest'] / 12 / 100;
    $after_rate    = $opt['after_interest'] / 12 / 100;

    $method        = $opt['method']; // annuity | equal_principal

    $grace_months  = $opt['grace_months'];

    // Hỗ trợ lãi suất
    $support_rate   = $opt['interest_support_rate'] / 100; 
    $support_months = $opt['interest_support_months'];

    // Tất toán sớm
    $early_fee_rate = $opt['early_payment_fee_rate'] / 100; 
    $settle_month   = $opt['settle_month']; // 0 = không tất toán
	
    // ==========================
    // Chuẩn bị biến
    // ==========================
    $remaining = $loan_amount;

    $interest_promo   = 0;
    $principal_promo  = 0;

    $interest_after   = 0;
    $principal_after  = 0;

    $monthly_promo = 0;
    $monthly_after = 0;
	
	// số tiền trả trước
	$down_payment = $total_price - $loan_amount;
	
	// lịch trả nợ
    $schedule = [];

    // ==========================
    // Nếu equal_principal -> gốc đều
    // ==========================
    $equal_principal = $method === "equal_principal"
        ? $loan_amount / $total_months
        : null;

    // ==========================
    // 1. Tính kỳ thanh toán ưu đãi
    // ==========================
    if ($method === "annuity") {
        // Dùng lãi suất ưu đãi nhưng áp dụng cho toàn kỳ vay
        $monthly_promo = $loan_amount * (
            $promo_rate * pow(1 + $promo_rate, $total_months)
        ) / (pow(1 + $promo_rate, $total_months) - 1);
    }
	
    for ($m = 1; $m <= $promo_months; $m++) {

        if ($settle_month > 0 && $m > $settle_month) break;

        // Lãi tháng này
        $interest = $remaining * $promo_rate;

        // Hỗ trợ lãi suất
        if ($m <= $support_months) {
            $interest *= (1 - $support_rate);
        }

        // Gốc
        if ($m <= $grace_months) {
            $principal = 0;
        } else {
            if ($method === "annuity") {
                $principal = $monthly_promo - $interest;
            } else {
                $principal = $equal_principal;
            }
        }

        $remaining -= $principal;
        $interest_promo  += $interest;
        $principal_promo += $principal;
		
		// Lưu lịch trả nợ
        $schedule[] = [
            "month"     => $m,
            "interest"  => $interest,
            "principal" => $principal,
            "payment"   => $interest + $principal,
            "remaining" => max($remaining, 0)
        ];
    }
    // Nếu tất toán trong giai đoạn ưu đãi → bỏ qua bước sau
    if ($settle_month > 0 && $settle_month <= $promo_months) {

        $early_fee_value = $remaining * $early_fee_rate;

        return [
            "method"            => $method,

            "total_price"       => $total_price,
            "loan_rate"         => $loan_rate,
            "loan_amount"       => $loan_amount,

            "monthly_promo"     => $monthly_promo,
            "monthly_after"     => 0,

            "interest_promo"    => $interest_promo,
            "principal_promo"   => $principal_promo,

            "interest_after"    => 0,
            "principal_after"   => 0,

            "total_interest"    => $interest_promo,
            "total_principal"   => $principal_promo,
            "total_payment"     => $interest_promo + $principal_promo + $early_fee_value,

            "remaining"         => max($remaining, 0),

            "early_fee_value"   => $early_fee_value
        ];
    }

    // ==========================
    // 2. Giai đoạn sau ưu đãi
    // ==========================
    $after_months = $total_months - $promo_months;
    if ($method === "annuity") {
        $monthly_after = $remaining * (
            $after_rate * pow(1 + $after_rate, $after_months)
        ) / (pow(1 + $after_rate, $after_months) - 1);
    }

    for ($i = 1; $i <= $after_months; $i++) {

        $current_month = $promo_months + $i;

        if ($settle_month > 0 && $current_month > $settle_month) break;

        $interest = $remaining * $after_rate;

        if ($method === "annuity") {
            $principal = $monthly_after - $interest;
        } else {
            $principal = $equal_principal;
        }

        $remaining -= $principal;
        $interest_after  += $interest;
        $principal_after += $principal;
		// Lưu lịch trả nợ
        $schedule[] = [
            "month"     => $current_month,
            "interest"  => $interest,
            "principal" => $principal,
            "payment"   => $interest + $principal,
            "remaining" => max($remaining, 0)
        ];
    }

    // ==========================
    // Tính phí tất toán nếu có
    // ==========================
    $early_fee_value = 0;

    if ($settle_month > 0) {
        $early_fee_value = $remaining * $early_fee_rate;
		// thêm bản ghi tất toán
        $schedule[] = [
            "month"     => $settle_month,
            "interest"  => 0,
            "principal" => $remaining,
            "payment"   => $remaining + $early_fee_value,
            "remaining" => 0
        ];
    }

    // ==========================
    // Tổng hợp trả về
    // ==========================
    return [
        "method"          => $method,

        "total_price"     => $total_price,
        "loan_rate"       => $loan_rate,
        "loan_amount"     => $loan_amount,
		'down_payment'	  => $down_payment,
		
        // Giai đoạn ưu đãi
        "monthly_promo"   => $monthly_promo,
        "interest_promo"  => $interest_promo,
        "principal_promo" => $principal_promo,

        // Giai đoạn sau ưu đãi
        "monthly_after"   => $monthly_after,
        "interest_after"  => $interest_after,
        "principal_after" => $principal_after,

        // Tổng
        "total_interest"  => $interest_promo + $interest_after,
        "total_principal" => $principal_promo + $principal_after,
        "total_payment"   => $interest_promo + $interest_after + $principal_promo + $principal_after + $early_fee_value,

        // Dư nợ cuối
        "remaining"       => max($remaining, 0),

        // Tất toán
        "early_fee_value" => $early_fee_value,
        "early_fee_rate"  => $early_fee_rate,
		'schedule' => $schedule,
    ];
}

            $principal = 0;
            $payment   = $interest;
        } else {
            if ($method === "equal_principal") {
                $principal = $equal_principal;
                $payment   = $principal + $interest;
            } else {
                $principal = $monthly_promo - $interest;
                $payment   = $monthly_promo;
            }
        }

        $remaining -= $principal;

        if ($m == 1) $first_payment = $payment;

        $interest_promo  += $interest;
        $principal_promo += $principal;

        $schedule[] = [
            "month"          => $m,
            "begin_balance"  => $begin_balance,
            "interest"       => $interest,
            "principal"      => $principal,
            "payment"        => $payment,
            "remaining"      => max($remaining, 0)
        ];
    }

    // Nếu tất toán trong giai đoạn ưu đãi
    if ($settle_month > 0 && $settle_month <= $promo_months) {

        $early_fee_value = $remaining * $early_fee_rate;

        return [
            "method"         => $method,
            "loan_amount"    => $loan_amount,
            "down_payment"   => $down_payment,

            "first_payment"  => $first_payment,

            "monthly_promo"  => $monthly_promo,
            "monthly_after"  => 0,

            "interest_promo"  => $interest_promo,
            "principal_promo" => $principal_promo,

            "total_interest"  => $interest_promo,
            "total_principal" => $principal_promo,
            "total_payment"   => $interest_promo + $principal_promo + $early_fee_value,

            "remaining"        => max($remaining, 0),
            "early_fee_value"  => $early_fee_value,

            "schedule"         => $schedule
        ];
    }

    // ==========================
    // Giai đoạn sau ưu đãi
    // ==========================
    $after_months = $total_months - $promo_months;

    if ($method === "annuity") {
        if ($after_rate > 0) {
            $monthly_after = $remaining *
                ( $after_rate * pow(1 + $after_rate, $after_months) ) /
                ( pow(1 + $after_rate, $after_months) - 1 );
        } else {
            $monthly_after = $remaining / $after_months;
        }
    }

    for ($i = 1; $i <= $after_months; $i++) {

        $current_month = $promo_months + $i;

        if ($settle_month > 0 && $current_month > $settle_month) break;

        $begin_balance = $remaining;

        $interest = $remaining * $after_rate;

        if ($method === "equal_principal") {
            $principal = $equal_principal;
            $payment   = $principal + $interest;
        } else {
            $principal = $monthly_after - $interest;
            $payment   = $monthly_after;
        }

        $remaining -= $principal;

        if ($current_month == 1) $first_payment = $payment;

        $interest_after  += $interest;
        $principal_after += $principal;

        $schedule[] = [
            "month"          => $current_month,
            "begin_balance"  => $begin_balance,
            "interest"       => $interest,
            "principal"      => $principal,
            "payment"        => $payment,
            "remaining"      => max($remaining, 0)
        ];
    }

    // ==========================
    // Tất toán sau ưu đãi
    // ==========================
    $early_fee_value = 0;

    if ($settle_month > 0) {

        $early_fee_value = $remaining * $early_fee_rate;

        $schedule[] = [
            "month"          => $settle_month,
            "begin_balance"  => $remaining,
            "interest"       => 0,
            "principal"      => $remaining,
            "payment"        => $remaining + $early_fee_value,
            "remaining"      => 0
        ];
    }

    // ==========================
    // Output
    // ==========================
    return [
        "method"           => $method,
        "loan_amount"      => $loan_amount,
        "down_payment"     => $down_payment,

        "first_payment"    => $first_payment,

        "monthly_promo"    => $monthly_promo,
        "monthly_after"    => $monthly_after,

        "interest_promo"   => $interest_promo,
        "principal_promo"  => $principal_promo,
        "interest_after"   => $interest_after,
        "principal_after"  => $principal_after,

        "total_interest"   => $interest_promo + $interest_after,
        "total_principal"  => $principal_promo + $principal_after,
        "total_payment"    => $interest_promo + $interest_after + $principal_promo + $principal_after + $early_fee_value,

        "remaining"        => max($remaining, 0),
        "early_fee_value"  => $early_fee_value,

        "schedule"         => $schedule
    ];
}
