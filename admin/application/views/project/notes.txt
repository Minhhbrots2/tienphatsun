    function loadImageToCanvas(imgUrl, callback) {
        var img = new Image();
        img.src = imgUrl;
        img.onload = function () {
            var canvas = document.createElement('canvas');
            canvas.width = img.width;
            canvas.height = img.height;
            var ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0);
            callback(canvas);
        };
    }

// Hàm tính toán khoảng cách giữa 2 điểm
function calculateDistance(p1, p2) {
    return Math.sqrt(Math.pow(p1.x - p2.x, 2) + Math.pow(p1.y - p2.y, 2));
}

// Hàm nhận diện và vẽ hình theo vùng có viền
function detectAndDraw(latlng) {
    var img = new Image();
    img.src = imageUrl;
    img.onload = function () {
        var canvas = document.createElement('canvas');
        var ctx = canvas.getContext('2d');
        canvas.width = img.width;
        canvas.height = img.height;
        ctx.drawImage(img, 0, 0);

        // Chuyển canvas thành ảnh xám để dễ dàng xử lý
        var imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
        var data = imgData.data;
        
        // Xử lý và nhận diện viền đen
        var contours = [];  // Mảng lưu trữ các đường viền nhận diện
        for (var i = 0; i < data.length; i += 4) {
            var r = data[i];
            var g = data[i + 1];
            var b = data[i + 2];

            // Kiểm tra nếu là màu đen (hoặc gần đen)
            if (r < 50 && g < 50 && b < 50) {
                // Tạo vùng viền tại các điểm đen
                var x = (i / 4) % canvas.width;
                var y = Math.floor((i / 4) / canvas.width);

                // Kiểm tra vùng xung quanh điểm này để xác định hình
                contours.push({ x: x, y: y });
            }
        }

        // Tìm đối tượng gần nhất với vị trí click
        var clickPoint = map.latLngToContainerPoint(latlng);
        var closestContour = null;
        var minDistance = Infinity;

        contours.forEach(function (point) {
            var distance = calculateDistance(clickPoint, point);
            if (distance < minDistance) {
                minDistance = distance;
                closestContour = point;
            }
        });

        // Nếu tìm thấy một điểm gần nhất, vẽ hình bao quanh
        if (closestContour) {
            // Giả sử rằng chúng ta nhận diện được hình chữ nhật hoặc các hình đơn giản
            var latlngs = [
                map.containerPointToLatLng(L.point(closestContour.x, closestContour.y)),
                map.containerPointToLatLng(L.point(closestContour.x + 100, closestContour.y)), // Xác định kích thước
                map.containerPointToLatLng(L.point(closestContour.x + 100, closestContour.y + 50)),
                map.containerPointToLatLng(L.point(closestContour.x, closestContour.y + 50))
            ];

            drawPolygon(latlngs);
        }
    };
}

map.on('dblclick', function (event) {
    var clickLatLng = event.latlng;
    detectAndDraw(clickLatLng); // Xử lý ảnh và vẽ hình bao quanh
});
